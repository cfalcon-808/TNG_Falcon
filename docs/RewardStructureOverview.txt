# ============================================================
#  Reward Structure Overview (env6.0)
#
#  Reward terms are controlled directly by DEFAULT_KNOBS.
#
#  To ablate or tune a term, override its knob key via reward_weights.
#  Example: reward_weights = {"REWARD_BUBBLE_SPACE": 0.0}
# ============================================================
#
# ------------------------------------------------------------
#  1. Terminal Outcomes (Sparse, Dominant Signals)
# ------------------------------------------------------------
#  Goat learner:
#    + REWARD_GOAT_WIN            when goats immobilize all tigers
#    + REWARD_TIGER_WIN           when tigers win (negative by default)
#
#  Tiger learner:
#    + REWARD_TIGER_WIN_BONUS     when tigers win
#    - REWARD_TIGER_LOSS_PENALTY  when tigers lose
#
#  Timeout modifiers:
#    MAX_TIMEOUT_SCALE and REPEAT_STALL_SCALE
#
# ------------------------------------------------------------
#  2. Step and Progress Shaping
# ------------------------------------------------------------
#  Base step term:
#    REWARD_STEP
#
#  Move-phase step ramp:
#    MOVE_STEP_BASE + MOVE_STEP_SLOPE * move_steps, clipped by MOVE_STEP_MIN
#
# ------------------------------------------------------------
#  3. Tactical and Positional Shaping
# ------------------------------------------------------------
#  Goat-facing shaping terms:
#    REWARD_GOAT_EATEN
#    REWARD_BLOCK_TIGER
#    NEAR_LOCK_BONUS
#    REWARD_BUBBLE_SPACE
#    REWARD_CLUSTER_TIGERS
#    REWARD_CENTER_GOAT
#    REWARD_CENTER_TIGER
#
#  Tiger-facing shaping terms:
#    REWARD_TIGER_CAPTURE
#    REWARD_BLOCK_TIGER
#    REWARD_CENTER_GOAT (used as magnitude in tiger center scoring)
#
# ------------------------------------------------------------
#  4. Anti-repeat and Invalid-action Penalties
# ------------------------------------------------------------
#    REWARD_REPEAT_STATE
#    MAX_REPEATS
#    REWARD_INVALID_SOFT
#    REWARD_INVALID_HARD
#
# ------------------------------------------------------------
#  5. Opponent Dynamics Knobs (Non-reward)
# ------------------------------------------------------------
#    BASE_TIGER_CAPTURE_BIAS
#
#  These do not directly add reward, but change opponent behavior.
#
# ------------------------------------------------------------
#  6. Ablation / Sweep Examples
# ------------------------------------------------------------
#  Disable bubble shaping:
#    reward_weights = {"REWARD_BUBBLE_SPACE": 0.0}
#
#  Reduce mobility shaping:
#    reward_weights = {"REWARD_BLOCK_TIGER": 0.04}
#
#  Soften repeat penalty:
#    reward_weights = {"REWARD_REPEAT_STATE": -0.2}
#
# ============================================================
