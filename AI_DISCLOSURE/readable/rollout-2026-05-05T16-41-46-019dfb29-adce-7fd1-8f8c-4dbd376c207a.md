Define policy helpers:

```python
def random_action(mask):
    legal = np.flatnonzero(mask)
    return int(np.random.choice(legal))


def make_model_policy(model_path, device="cpu", deterministic=True):
    model = MaskablePPO.load(model_path, device=device)

    def policy(obs, mask):
        obs_arr = np.asarray(obs).reshape(1, -1)
        mask_arr = np.asarray(mask, dtype=bool).reshape(1, -1)
        action, _ = model.predict(
            obs_arr,
            deterministic=deterministic,
            action_masks=mask_arr,
        )
        return int(action[0])

    return policy
```

Then define your episode collector. Important detail: collect the state **before** the action, because that is the state the policy is evaluating.

```python
def collect_episode(env, goat_policy, source_name):
    obs, info = env.reset()

    episode_rows = []
    done = False
    move_index = 0

    while not done:
        mask = env.get_action_mask()

        row = {
            "source": source_name,
            "move_index": move_index,
            "state": obs.copy(),
            "board": obs[:23].copy(),
            "goats_eaten": int(obs[23]),
            "phase": int(obs[24]),
            "goats_placed": int(env.goats_placed),
            "turn_counter": int(env.turns),
            "legal_action_count": int(mask.sum()),
        }

        action = goat_policy(obs, mask)

        obs, reward, terminated, truncated, info = env.step(action)
        done = terminated or truncated

        episode_rows.append(row)
        move_index += 1

    winner = info.get("winner")
    total_moves = len(episode_rows)

    if winner == "Goat":
        outcome_class = "goat_win"
        value_label = 1.0
    elif winner == "Tiger":
        outcome_class = "tiger_win"
        value_label = 0.0
    else:
        outcome_class = "timeout"
        value_label = 0.5

    for row in episode_rows:
        row["winner"] = winner
        row["outcome_class"] = outcome_class
        row["value_label"] = value_label
        row["total_moves"] = total_moves
        row["progress_ratio"] = row["move_index"] / max(total_moves - 1, 1)

    return episode_rows, outcome_class
```

```markdown
## 5. Episode Collection Function
```

This is the most important part.

The pattern should be:

1. Reset env.
2. Save state before each action.
3. Step environment.
4. Stop at terminal/truncated.
5. Apply final outcome label to every saved row.

```python
def classify_outcome(winner):
    if winner == "Goat":
        return "goat_win", 1.0
    if winner == "Tiger":
        return "tiger_win", 0.0
    return "timeout", 0.5
```

Episode collector:

```python
def collect_episode(source, episode_id):
    env = TnGEnv(**source["env_kwargs"])
    policy = source["policy"]

    obs, info = env.reset()

    rows = []
    done = False
    move_index = 0

    while not done:
        mask = env.get_action_mask()

        rows.append({
            "episode_id": episode_id,
            "source": source["name"],
            "move_index": move_index,
            "state": obs.copy(),
            "board": obs[:23].copy(),
            "goats_eaten": int(obs[23]),
            "phase": int(obs[24]),
            "goats_placed": int(env.goats_placed),
            "turn_counter": int(env.turns),
            "legal_action_count": int(mask.sum()),
            "tiger_ai": source["env_kwargs"].get("tiger_ai"),
        })

        action = policy(obs, mask)

        obs, reward, terminated, truncated, info = env.step(action)
        done = terminated or truncated
        move_index += 1

    winner = info.get("winner")
    outcome_class, value_label = classify_outcome(winner)
    total_moves = len(rows)

    for row in rows:
        row["winner"] = winner
        row["outcome_class"] = outcome_class
        row["value_label"] = value_label
        row["total_moves"] = total_moves
        row["progress_ratio"] = row["move_index"] / max(total_moves - 1, 1)

    return rows, outcome_class
```
